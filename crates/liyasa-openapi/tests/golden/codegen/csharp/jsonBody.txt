using System.Net.Http;

using var client = new HttpClient();
using var request = new HttpRequestMessage(new HttpMethod("POST"), "https://api.example.com/v1/notes");
request.Content = new StringContent("{\n  \"text\": \"It's here\",\n  \"pinned\": true,\n  \"tags\": [\n    \"new\"\n  ]\n}", System.Text.Encoding.UTF8, "application/json");

var response = await client.SendAsync(request);
Console.WriteLine(await response.Content.ReadAsStringAsync());
