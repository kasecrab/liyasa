using System.Net.Http;

using var client = new HttpClient();
using var request = new HttpRequestMessage(new HttpMethod("POST"), "https://api.example.com/v1/subscribe");
request.Content = new StringContent("email=user%40example.com&plan=free", System.Text.Encoding.UTF8, "application/x-www-form-urlencoded");

var response = await client.SendAsync(request);
Console.WriteLine(await response.Content.ReadAsStringAsync());
