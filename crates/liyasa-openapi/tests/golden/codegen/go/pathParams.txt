package main

import (
	"fmt"
	"io"
	"net/http"
)

func main() {
	request, err := http.NewRequest("GET", "https://api.example.com/v1/users/u_1/notes/7", nil)
	if err != nil {
		panic(err)
	}

	response, err := http.DefaultClient.Do(request)
	if err != nil {
		panic(err)
	}
	defer response.Body.Close()

	payload, err := io.ReadAll(response.Body)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(payload))
}
