import httpx

url = "https://api.example.com/v1/notes"
headers = {
    "Content-Type": "application/json",
}
payload = {
    "text": "It's here",
    "pinned": True,
    "tags": [
        "new"
    ]
}

with httpx.Client() as client:
    response = client.request("POST", url, headers=headers, json=payload)
    print(response.json())
